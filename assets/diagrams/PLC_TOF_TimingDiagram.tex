\begin{circuitikz}[
    american,
    voltage dir=RP,
    >=latex,
    %>={Latex[length=5mm, width=2mm]},
    alt={Diagram showing a timing diagram over an 18-second timeline illustrating the digital state transitions for an PLC instruction's input signal alongside its associated status bits: Enable (EN), Timer Timing (TT), and Done (DN).}
    ]
    
    %============================================================================================
    % Draw and label the axes
    %============================================================================================

    % Axis
    \draw[
        BakerBlack
    ] 
    (0,8.5) 
    -- (0,0) 
    -- (18.5,0);

    % X axis grid and labels
    \foreach \x [count=\i] in {2,4,...,18}
    {
        \draw[
            Iron1,
            ultra thin,
            dashed
        ] 
        (\x,0) 
        -- (\x,8.5);

        \node[
            text=BakerBlack,
            anchor=north,
            align=center
        ]
        at (\x,0)
        {{\i}};
    }

    % X-axis label
    \node[BakerBlack, anchor=center, align=center] at (9.25,-1){Seconds};

    % Y axis grid and labels
    \foreach \y in {1,2,...,8}
    {
        % Evaluates to 1 for even y, 0 for odd y
        \pgfmathsetmacro{\val}{int(mod(\y + 1, 2))}

        \draw[
            Iron1,
            ultra thin,
            dashed
        ] 
        (0,\y) 
        -- (18.5,\y);

        \node[
            BakerBlack, 
            anchor=east, 
            align=right
        ] 
        at (0,\y){\val};
    }

    %============================================================================================
    % Individual Bit Lines
    %============================================================================================

    % DN bit
    \node[BakerBlack, anchor=east, align=right] at (-0.5,1.5){DN};
    \draw[BakerADARed, ultra thick] (0,2) -- (12,2) -- (12,1) -- (14,1) -- (14,2) -- (18,2);

    % TT bit
    \node[BakerBlack, anchor=east, align=right] at (-0.5,3.5){TT};
    \draw[BakerADARed, ultra thick] (0,3) -- (4,3) -- (4,4) -- (6,4) -- (6,3) -- (8,3) -- (8,4) -- (12,4) -- (12,3) -- (18,3);

    % EN bit
    \node[BakerBlack, anchor=east, align=right] at (-0.5,5.5){EN};
    \draw[BakerADARed, ultra thick] (0,5) -- (2,5) -- (2,6) -- (4,6) -- (4,5) -- (6,5) -- (6,6) -- (8,6) -- (8,5) -- (14,5) -- (14,6) -- (18,6);

    % Input
    \node[BakerBlack, anchor=east, align=right] at (-0.5,7.5){Input};
    \draw[BakerADARed, ultra thick] (0,7) -- (2,7) -- (2,8) -- (4,8) -- (4,7) -- (6,7) -- (6,8) -- (8,8) -- (8,7) -- (14,7) -- (14,8) -- (18,8);
    
\end{circuitikz}
