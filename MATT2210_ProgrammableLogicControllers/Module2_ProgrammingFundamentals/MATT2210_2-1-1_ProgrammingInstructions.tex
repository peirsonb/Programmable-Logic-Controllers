\section{Data Types}

\begin{frame}
    \frametitle{BIT}

    \begin{itemize}
        \item 1 bit
        \item The smallest unit of memory
        \item Represents discrete binary states
        \begin{itemize}
            \item 0 (OFF/FALSE)
            \item 1 (ON/TRUE)
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{NIBBLE}

    \begin{itemize}
        \item 4 bits
        \item 1 binary coded decimal (BCD) digit
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{BYTE}

    \begin{itemize}
        \item 8 Bits
        \item 256 unique combinations
        \item Typically used for ASCII characters--1 BYTE = 1 character
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{WORD}

    \begin{itemize}
        \item 16 Bits
        \item Historical standard register size in PLCs (AB SLC 500/Siemens S7-300)
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{DWORD/LWORD}

    \begin{itemize}
        \item DWORD--32 bits
        \item LWORD--64 bits
        \item Modern architecture
        \item Used for high precision motion/math in older systems
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Sub-Bit Access}

    \begin{itemize}
        \item These data types are most often used as a number
        \item But they are \emph{also} just a collection of bits
        \item We can access the individual bit inside a word using \emph{dot notation}
        \begin{itemize}
            \item \texttt{Word\_Var.0}, \texttt{Word\_Var.7}, etc.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Signed Integers}
    
    \begin{itemize}
        \item Integers can be signed (+/-) or unsigned
        \item The defautl in most PLCs is signed--unsigned is optional
        \item In a signed integer the first bit is the sign: 0 = +, 1 = -
        \item Because we lose a bit, the capacity of the number is effectively cut in half
        \item \texttt{INT}: -32,768 to 32,767
        \item \texttt{UINT}: 0 to 65,535
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Floating Point Numbers}
    
    \begin{itemize}
        \item IEEE 754 Format
        \item 32-bits: 1 sign, 8 exponent bits, 23 mantissa bits
        \item An integer is \emph{only} whole numbers, \texttt{REAL} must be used if a decimal point is necessary
        \item \emph{NEVER} use an equality check (\texttt{REAL\_1 = 10.0}) on a \texttt{REAL}--floating point numbers have in-built rounding errors
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Addressing Memory in PLCs}

    \begin{itemize}
        \item \textbf{Direct Memory Addressing:} Logic references explicit \emph{physical} register locations in the CPU
        \item \textbf{Tag-Based Addressing:} Programmer creates descriptive, symbolic names stored in a database
        \item IEC 61131-3 standardizes both approaches
        \item Older PLCs pretty exclusively direct addressed, modern PLCs mostly use tag-based databases
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Take Caution with Latches}

    \begin{itemize}
        \item Be \emph{very} careful with latches--my recommendation is to not use them as much as possible
        \item OTE is \emph{volatile}--when power cycles the output is set to 0 (off)
        \item OTL is \emph{NOT} volatile--if it's on when power goes out it will be on when power is restored
        \item This can and has caused unwanted/unsafe machine motion
    \end{itemize}
\end{frame}

\section{Timers and Counters}

\begin{frame}
    \frametitle{Timers and Counters}

    \begin{itemize}
        \item Timers time, counters count--but their under-the-hood structure is \emph{very} similar
        \item ``Timer'' and ``counter'' refer to the \emph{variable type}
        \item The timer or counter variable is then used in conjunction with a timer or counter instruction
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Timer and Counter Variable Types}

    \begin{itemize}
        \item Timers and counters are made up of 3 words:
        \begin{enumerate}
            \item Word 0: Control word--used as bits, what the bits \emph{do} vary between timers and counters
            \item Word 1: Preset (PRE)--the time/count target
            \item Word 2: Accumulated (ACC)--the current time/count
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Timer Variables}

    \begin{description}
        \item[EN] Rung logic before the instruction is true
        \item[TT] The timer is actively timing
        \item[DN] The preset value has been reached
        \item[PRE] The target time, in \emph{milliseconds}
        \item[ACC] The current timer value, in \emph{milliseconds} 
    \end{description}
\end{frame}

\begin{frame}
    \frametitle{Time Base}

    \begin{itemize}
        \item Some platforms use a timer time base
        \item Time base $\times$ preset = total time
        \item AB RSlogix 500: time base can be 0.01s or 1s
        \begin{itemize}
            \item Base = 0.01, Preset = 500, Time = 5s
        \end{itemize}
        \item AB RSLogix 5000: time base is \emph{always} 0.001s
        \begin{itemize}
            \item Base = 0.001, Preset = 500, Time = 0.5s
        \end{itemize} 
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Timer Variable Warning}

    \begin{itemize}
        \item Timer variables can only be used \emph{once}
        \item Software will \emph{allow} you to use them more than once, but you will definitely see fun and interesting behavior
        \item One timer variable per timer instruction
    \end{itemize}
\end{frame}

\subsection{Counters}

\begin{frame}
    \frametitle{Counter Variables}

    \begin{description}
        \item[CU] The logic before the \texttt{CTU} instruction is true
        \item[CD] The logic before the \texttt{CTD} instruction is true
        \item[DN] The preset value has been reached
        \item[OV] The \texttt{ACC} has exceeded the DINT limit of 2,147,483,647
        \item[UN] The \texttt{ACC} has exceed the DINT limit of -2,147,483,648 
        \item[PRE] The target count
        \item[ACC] The current count
    \end{description}
\end{frame}

\begin{frame}
    \frametitle{Counter Usage}

    \begin{itemize}
        \item Unlike timer variables, counter variables can be used \emph{twice}
        \item Once for a count up (CTU) and once for a count down (CTD)
        \item Recommendation: put the CTU and CTD \emph{very} close to one another--makes it easier to troubleshoot
    \end{itemize}
\end{frame}

\subsection{Count Up}

\begin{frame}
    \frametitle{Count Up (CTU)}

    \begin{columns}
        \begin{column}{0.48\linewidth}
            \begin{itemize}
                \item Adds one to the count when the preceding logic is true
                \item \emph{ONLY} adds one--the logic must go false in order to count again
                \item If the counter variable needs to be cleared, must use a RES instruction
            \end{itemize}
        \end{column}

\begin{frame}
    \frametitle{Math Instruction Arguments}

    \begin{itemize}
        \item The instructions on the previous slide accept 3 arguments: Source A, Source B, and Destination
        \item Source A and Source B can be either a variable (tag) or a fixed number
        \item Destination \emph{must} be a variable--the instruction will store the result here
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Argument Example}

    \begin{figure}
        \centering
        \input{../../assets/diagrams/PLC_MathInstructions_Arguments.tex}

        \vspace{0.5em}
        {\footnotesize The sources can be fixed numbers or variables.\par}
    \end{figure}
\end{frame}

\begin{frame}
    \frametitle{Math Instruction Data Types}

    \begin{itemize}
        \item Math instructions work on all numeric data types
        \item Data types \emph{do not} need to match
        \item Be careful--if the math would produce a decimal, but the variable is an integer all decimal points will be chopped off
        \item Can be useful if you don't want the decimals
    \end{itemize}
\end{frame}

    \frametitle{Floating Point Example}

    \end{figure}
\end{frame}

\begin{frame}
    \frametitle{Math Instruction Behavior}

    \begin{itemize}
        \item Source A comes first in the operation--Source A minus Source B, Source A divided by Source B, etc
        \item The destination variable \emph{can be} the same variable as Source A--useful for a running count
    \end{itemize}
