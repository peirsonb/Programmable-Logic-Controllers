\begin{frame}
    \frametitle{Bad Logic Compounds Downtime}

    \begin{itemize}
        \item Extended Mean Time to Repair (MTR): Up to 80\% of downtime is spent deciphering code rather than fixing hardware
        \item Troubleshooting Friction: Intermingled safety permissives, sequence logic, and direct I/O forces trial-and-error debugging
        \item Modification Risk: Fixing one rung accidentally breaks secondary machine routines
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Spaghetti Code vs. Architecture}

    \begin{columns}
        \begin{column}{0.48\linewidth}
            \begin{block}{Unstructured Logic}
                Direct physical I/O scattered across dozens of subroutines with unbuffered signals and nested latches
            \end{block}
        \end{column}
        \begin{column}{0.48\linewidth}
            \begin{block}{Layered Architecture}
                Clean separation between Input Mapping, State Machines, Output Buffers, and Alarm Routines.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Micro-Level vs. Macro-Level}

    \begin{itemize}
        \item High-quality industrial code requires mastery at both the instruction level and architectural level
        \item Micro (Rung Mechanics): How individual instructions, logic gates, and execution operate within a single scan cycle
        \item Macro (Program Layout): How routines, modules, are organized across the entire controller
        \item Flawless micro-logic fails inside a disorganized architecture; brilliant macro-architecture is useless if individual rungs are buggy
    \end{itemize}
\end{frame}
